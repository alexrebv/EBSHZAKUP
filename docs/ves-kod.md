```js
  'FBS остатки Ламода': ямLm.лмОстаткиFbs,

  'Заказы Ламода FBS': ямLm.лмЗаказыFbs,
  // у М.Видео публичного API продавца нет: эти листы заполняются вручную
};

async function выполнить(задача) {
  const обработчик = ОБРАБОТЧИКИ[задача.имя];
  if (!обработчик) {
    await отметитьЗапуск(задача, 'нет коннектора, данные заливаются вручную');
    return;
  }
  const старт = Date.now();
  try {
    const результат = await обработчик(задача);
    const сек = ((Date.now() - старт) / 1000).toFixed(1);
    // период показываем только у задач, которые его используют: у остатков он смысла не несёт
    const подпись = задача.имя.startsWith('Заказы') ? описаниеПериода(задача) : '';
    const хвост = подпись ? `, ${подпись}` : '';
    await отметитьЗапуск(задача, `${результат}, ${сек} c${хвост}`);
    console.log(`[ok] ${задача.имя}: ${результат} (${сек} c)`);
  } catch (е) {
    await отметитьЗапуск(задача, `ОШИБКА: ${е.message}`.slice(0, 400));
    console.error(`[fail] ${задача.имя}: ${е.message}`);
  }
}

/**
 * Проходы нельзя пускать внахлёст. Задача может идти дольше минуты, а таймер
 * тикает всё равно: тогда «Последний запуск» ещё не записан, `пора()` снова
 * говорит «да», и та же задача уходит в работу вторым и третьим экземпляром.
 */
let идёт = false;

async function проход(принудительно = null) {
  const список = await задачи();
  const кЗапуску = принудительно
    ? список.filter((з) => з.имя === принудительно)
    : список.filter(пора);
  if (принудительно && !кЗапуску.length) {
    console.error(`задача «${принудительно}» не найдена в блоке задач на листе настроек`);
    process.exitCode = 1;
    return;
  }
  if (!кЗапуску.length) return;
  console.log(`к запуску: ${кЗапуску.map((з) => з.имя).join(', ')}`);
  for (const задача of кЗапуску) {
    await выполнить(задача);
  }
}

const аргументы = process.argv.slice(2);
const одинРаз = аргументы.includes('--once');
const позиция = аргументы.indexOf('--task');
const однаЗадача = позиция >= 0 ? аргументы[позиция + 1] : null;

/** Тик расписания: пропускаем, если прошлый проход ещё не закончился. */
function тик() {
  if (идёт) {
    console.log('прошлый проход ещё идёт, тик пропущен');
    return;
  }
  идёт = true;
  // проход роняет процесс только в разовых режимах: воркер должен пережить
  // временную недоступность Таблиц и попробовать снова через минуту
  проход()
    .catch((е) => console.error('сбой прохода:', е.message))
    .finally(() => { идёт = false; });
}

if (однаЗадача) {
  await проход(однаЗадача);
} else if (одинРаз) {
  await проход();
} else {
  console.log('сервис запущен, проверка расписания раз в минуту');
  тик();
  setInterval(тик, 60000);
}
```

```js
import { НАСТРОЙКИ, ЗАДАЧИ } from './schemas.js';
import { читать, писать } from './sheets.js';
import { вТаблицу, изТаблицы } from './http.js';

const ЛИСТ = НАСТРОЙКИ.лист;

/** Ключи берём из окружения — на листе их держать не обязательно. */
export const доступ = (имя) => process.env[имя] || '';

export function требовать(...имена) {
  const нет = имена.filter((и) => !доступ(и));
  if (нет.length) throw new Error(`не заполнено: ${нет.join(', ')}`);
  return имена.map(доступ);
}

export async function параметры() {
  const строки = await читать(
    `'${ЛИСТ}'!A${НАСТРОЙКИ.параметрыОт}:B${НАСТРОЙКИ.параметрыДо}`,
  );
  const карта = {};
  for (const [имя, знач] of строки) {
    if (имя) карта[String(имя).trim()] = знач;
  }
  return карта;
}

/**
 * Строка блока настроек → задача. Вынесено отдельно и без обращений к книге:
 * разбор колонок — самое хрупкое место, и проверять его надо без сети.
 */
export function задачаИзСтроки(р, строка) {
  // в блоке возвратов обе колонки, D и E, озаглавлены «С даты», и неизвестно,
  // в какую вписана дата. Принимаем из любой: настоящая дата в колонке глубины
  // глубиной быть не может, а глубина датой не притворится — год не тот
  // берём ту колонку, которая разбирается в настоящую дату, а не первую непустую:
  // пробел, ноль или оставшаяся подпись в E иначе отбросили бы дату из D.
  // Если ни одна не разбирается, отдаём E как есть — пусть границаИстории
  // скажет понятное «не разобрана «С даты»», а не тихо загрузит весь список
  const датаВГлубине = датойПохоже(р[3]);
  const сдаты = датойПохоже(р[4]) ? р[4] : (датаВГлубине ? р[3] : (р[4] ?? ''));
  const отметка = изТаблицы(р[5]);
  const результат = String(р[6] ?? '').trim();
  return {
    имя: String(р[0] ?? '').trim(),
    строка,
    включена: Number(р[1]) === 1,
    каждые: Number(р[2]) || 30,
    глубина: датаВГлубине ? 0 : глубинаИз(р[3]),
    // значение ячейки держим сырым: дата из книги приходит числом,
    // разбирать её умеет только изТаблицы
    сдаты,
    сдатыИзГлубины: датаВГлубине && !датойПохоже(р[4]),
    последний: отметка,
    // «Последний запуск» отмечает попытку, а не успех: отметитьЗапуск пишет её
    // и при ошибке. Для периода нужен именно успех, иначе упавший прогон сдвинет
    // окно вперёд и невыгруженные заказы будут потеряны навсегда.
    // Чтобы метка успеха пережила ошибку, она дописывается в текст «Результата».
    успешный: успехИз(результат, отметка),
  };
}

export async function задачи() {
  // дочитываем запас ниже блока: строка за его концом не работает, но и молчать
  // о ней нельзя — иначе вписанная в книгу задача просто не запустится
  const строки = await читать(
    `'${ЛИСТ}'!A${НАСТРОЙКИ.задачиОт}:G${НАСТРОЙКИ.задачиДо + НАСТРОЙКИ.задачиЗапас}`,
  );
  const итог = [];
  строки.forEach((р, i) => {
    const номер = НАСТРОЙКИ.задачиОт + i;
    const имя = String(р[0] ?? '').trim();
    if (номер > НАСТРОЙКИ.задачиДо) {
      if (ЗАДАЧИ.includes(имя)) {
        console.warn(`строка ${номер}: «${имя}» ниже блока задач (до ${НАСТРОЙКИ.задачиДо}) `
          + 'и не запустится — поднимите задачиДо в src/schemas.js');
      }
```

```js
  // эти три часа терялись навсегда

 * Период выгрузки. Две настройки работают вместе, а не вместо друг друга:
 *
 * «С даты» — граница истории. Раньше неё не ходим никогда, и с неё же идёт
 * первичная заливка, пока «Последний запуск» пуст. Очистить «Последний запуск» —
 * значит заново залить всю историю с этой даты.
 *
 * «Глубина дней» — скользящее окно, которое обновляется каждый прогон: свежие
 * заказы добавляются, изменения в уже записанных подхватываются.
 *
 * Если сервис простоял дольше глубины, окно растягивается до прошлого запуска,
 * иначе заказы за простой были бы потеряны навсегда.
 */
export function период(задача) {

  // водяной знак — последний УСПЕШНЫЙ прогон; после ошибки окно растянется назад
  // и заберёт то, что не удалось выгрузить
  const знак = задача.успешный || null;

  // нельзя: первичная заливка считалась бы состоявшейся и история не пришла бы
  // никогда. Проверка стоит до обеих веток — в заливке дата берётся как есть
  const негодна = (с) => с.getTime() > по.getTime();
  const отказ = (с) => new Error(`«С даты» ${вТаблицу(с).slice(0, 10)} позже сегодняшнего дня`);

  if (граница && !знак) {

    return { с: граница, по, режим: 'заливка', историяГлубжеОкна: false, растянутоЗнаком: false };

  // сужает окно под предел своего метода, по этому признаку отличает дыру в

  let растянутоЗнаком = false;
```

```js
  const буква = колонка(колонок);
  if (Math.max(доСтроки, 2) < отСтроки) return;
  await api.spreadsheets.values.clear({
    spreadsheetId: SHEET_ID,
    range: `'${лист}'!A${отСтроки}:${буква}${Math.max(доСтроки, 2)}`,
  });
}

/** Ячейка с датой в книге лежит числом, а мы пишем её текстом — приводим к одному виду. */
const ДАТА_ТЕКСТОМ = /^\d{1,2}\.\d{1,2}\.\d{4}(?: \d{1,2}:\d{2}(?::\d{2})?)?$/;
/** Шестнадцать цифр и больше: столько Таблицы и Number уже не держат без потерь. */
const ДЛИННОЕ_ЦЕЛОЕ = /^\d{16,}$/;
const ЭПОХА_ТАБЛИЦ = Date.UTC(1899, 11, 30);

function сравнимо(значение) {
  if (значение === null || значение === undefined) return '';
  // апостроф — это пометка «считать текстом», а не часть значения: Таблицы его
  // снимают при записи, и сравнивать надо уже без него
  const текст = String(значение).trim().replace(/^'/, '');
  if (текст === '') return '';
  if (ДАТА_ТЕКСТОМ.test(текст)) {
    const д = изТаблицы(текст);
    if (д) {
      const мс = Date.UTC(д.getFullYear(), д.getMonth(), д.getDate(),
        д.getHours(), д.getMinutes(), д.getSeconds());
      return `#${Math.round((мс - ЭПОХА_ТАБЛИЦ) / 1000)}`;
    }
  }
  // Целое длиннее пятнадцати цифр оставляем текстом. `Number` держит пятнадцать
  // значащих цифр, а `srid` бывает девятнадцатизначным: пять разных заказов
  // свелись бы к одному ключу, записались одной строкой, и на следующем прогоне
  // новый заказ затёр бы чужую. Числовое приведение тут ничего и не чинило —
  // в обе стороны такое число всё равно не ходит
  if (ДЛИННОЕ_ЦЕЛОЕ.test(текст)) return текст;
  // числа и числовые строки сводим к одному виду: штрихкод может вернуться числом,
  // цена «100.00» — числом 100, дата — серийным номером книги
  const ч = Number(текст.replace(',', '.'));
  if (Number.isFinite(ч)) return `#${Math.round(ч * 86400)}`;
  return текст;
}

const безПометки = (з) => String(з ?? '').trim().replace(/^'/, '');

/**
 * Строки совпадают, если различие только в «Обновлено» — она всегда последняя.
 *
 * Сравнение ПАРНОЕ, и порядок доводов важен: «книжная» пришла из книги,
 * «наша» собрана коннектором. Там, где мы пометили значение как текст, числовое
 * приведение применять нельзя: книжное 12345 и наше «0012345» свелись бы к
 * одному числу, и испорченный артикул не починился бы никогда. Для помеченных
 * значений сравниваем текст с текстом; для остальных — как раньше, иначе SKU,
 * который мы шлём строкой, а книга отдаёт числом, вечно считался бы изменённым.
 */
function одинаковы(книжная, наша, ширина) {
  for (let i = 0; i < ширина - 1; i += 1) {
    const н = наша[i];
    if (typeof н === 'string' && н.startsWith("'")) {
      if (безПометки(н) !== безПометки(книжная[i])) return false;
    } else if (сравнимо(книжная[i]) !== сравнимо(н)) return false;
  }
  return true;
}

/** Соседние номера строк сливаем в один диапазон: меньше запросов к API. */
function вДиапазоны(номера) {
  const куски = [];
  for (const н of [...номера].sort((x, y) => x - y)) {
    const последний = куски[куски.length - 1];
    if (последний && н === последний[последний.length - 1] + 1) последний.push(н);
    else куски.push([н]);
  }
  return куски;
}

async function писатьПачкой(данные) {
  if (!данные.length) return;
  const api = await таблицы();
  await api.spreadsheets.values.batchUpdate({
    spreadsheetId: SHEET_ID,
    requestBody: { valueInputOption: 'USER_ENTERED', data: данные },

/**
 * Пишет строки в лист. Остатки перезаписываются целиком, заказы дозаписываются:
 * строка с тем же ключом обновляется, новая уходит в конец. Строки за пределами
 * периода не трогаются — поэтому сузить «Глубина дней» безопасно, накопленная
 * история никуда не денется.
 *
 * Не изменившиеся строки не переписываются вообще. Иначе каждый прогон гонял бы
 * в Таблицы весь лист целиком ради одной только колонки «Обновлено»: на листе
 * с 17 тысячами заказов это две сотни тысяч ячеек за прогон.
 *
 * Ширина записи ограничена схемой — служебные колонки книги остаются нетронутыми.
 */
export async function сохранить(лист, строки) {
  const схема = СХЕМЫ[лист];
  if (!схема) throw new Error(`нет схемы для листа «${лист}»`);
  const ширина = схема.length;
  const буква = колонка(ширина);

  /**
   * Заголовки дозаполняем ровно в пустые клетки первой строки и отдельными
   * диапазонами: занятую клетку не трогаем вовсе — там может стоять своя
   * формулировка или формула, а переписать её значением значило бы её потерять.
   */
  const правкиШапки = (шапка) => {
    const пустые = схема.map((_, i) => i).filter((i) => !String(шапка[i] ?? '').trim());
    return вДиапазоны(пустые).map((к) => ({
      range: `'${лист}'!${колонка(к[0] + 1)}1`,
      values: [к.map((i) => схема[i])],
    }));
  };

  if (СНИМКИ.has(лист)) {
    if (!строки.length) {
      // очистка снимка идёт до записи, поэтому пустой ответ стёр бы весь лист
      // и отметился бы в книге как успешный прогон
      throw new Error(`«${лист}»: пришло 0 строк, снимок не перезаписан — `
        + 'прежние остатки в книге сохранены');
    }
    // сначала пишем, потом убираем хвост: обрыв между шагами оставит лист
    // с данными, а не пустым. Границу берём по числу новых строк, а не по
    // колонке A: в остатках ВБ там штрихкод, а он бывает пустым
    const шапка = (await читать(`'${лист}'!A1:${буква}1`))[0] || [];
    const правки = правкиШапки(шапка);
    await писать(`'${лист}'!A2`, строки);
    if (правки.length) await писатьПачкой(правки);
    await очистить(лист, ширина, ПРЕДЕЛ, строки.length + 2);
    return {
      всего: строки.length, новых: строки.length, изменено: строки.length,
      заметка: правки.length ? ', шапка проставлена' : '',
    };
  }

  const ключи = КЛЮЧИ[лист] || [0];
  // ключ считаем тем же приведением, что и сравнение строк: книга возвращает
  // значения уже по-своему — штрихкод числом, «0468…» без ведущего нуля, — и
  // сырое сравнение не нашло бы такую строку в указателе, дописывая её заново
  // при каждом прогоне, пока лист не упрётся в предел
  const ключОт = (строка) => ключи.map((j) => сравнимо(строка[j])).join('|');
  // читаем с первой строки: шапка приезжает тем же запросом, лишних не надо
  const всё = await читать(`'${лист}'!A1:${буква}${ПРЕДЕЛ}`);
  const шапка = всё[0] || [];
  const было = всё.slice(1);
  const индекс = new Map();
  было.forEach((строка, i) => {
    const к = ключОт(строка);
    if (к.replace(/\|/g, '')) индекс.set(к, i);
  });

  if (было.length >= ПРЕДЕЛ - 1) {
    // дальше по листу есть строки, которых мы не видели: их ключи не найдутся,
    // а дозапись ляжет поверх них. Лучше честно упасть, чем испортить историю
    throw new Error(`лист «${лист}» дочитан до предела ${ПРЕДЕЛ} строк: `
      + 'перенесите старые заказы в архивный лист, дозапись остановлена');
  }

  const итог = было.map((с) => [...с]);
  const правленые = new Set();
  let новых = 0;
  let безКлюча = 0;
  for (const строка of строки) {
    const к = ключОт(строка);
    // строка без ключа не находится при следующем прогоне и дозаписывалась бы
    // заново каждый раз, пока лист не упрётся в предел
    if (!к.replace(/\|/g, '')) { безКлюча += 1; continue; }
    if (индекс.has(к)) {
      const i = индекс.get(к);
      if (одинаковы(итог[i], строка, ширина)) continue;
      итог[i] = строка;
      if (i < было.length) правленые.add(i);
    } else {
      индекс.set(к, итог.length);
      итог.push(строка);
      новых += 1;
    }
  }

  if (итог.length + 1 > ПРЕДЕЛ) {
    // дозапись ушла бы ниже окна чтения, и следующий прогон этих строк не увидит
    throw new Error(`«${лист}»: со свежими строками выходит ${итог.length} при пределе `
      + `${ПРЕДЕЛ} — перенесите старые заказы в архивный лист, запись остановлена`);
  }

  let заметка = '';
  if (итог.length + 1 > ПРЕДЕЛ * 0.9) {
    заметка = `, строк ${итог.length} при пределе ${ПРЕДЕЛ} — пора в архив`;
    console.warn(`[${лист}]${заметка}`);
  }
  if (безКлюча) заметка += `, без ключа пропущено ${безКлюча}`;

  // в книгу уходят ТОЛЬКО наши строки. Отправить обратно прочитанное из книги
  // нельзя: апостроф-пометка в ответ не возвращается, и правильный текстовый
  // артикул при обратной записи снова разобрался бы как дата, а формула
  // в колонке схемы затёрлась бы своим же вычисленным значением
  const данные = вДиапазоны(правленые).map((к) => ({
    range: `'${лист}'!A${к[0] + 2}`,
    values: к.map((i) => итог[i]),
  }));
  if (новых) {
    данные.push({ range: `'${лист}'!A${было.length + 2}`, values: итог.slice(было.length) });
  }
  const правкиШ = правкиШапки(шапка);
  данные.push(...правкиШ);
  if (правкиШ.length) заметка += ', шапка проставлена';

  for (let i = 0; i < данные.length; i += ДИАПАЗОНОВ_ЗА_РАЗ) {
    await писатьПачкой(данные.slice(i, i + ДИАПАЗОНОВ_ЗА_РАЗ));
  }

  return { всего: итог.length, новых, изменено: правленые.size, заметка };
}
```

```js
            артикул: к.vendorCode || '-',
            nmID: к.nmID,
            chrtID: размер.chrtID,
          });
        }
      }
    }
    const к = ответ.cursor || {};
    if (карточки.length < 100 || !к.updatedAt) break;
    курсор = { limit: 100, updatedAt: к.updatedAt, nmID: к.nmID };
    await сон(300);
  }
  карточки = { время: Date.now(), список: итог };
  return итог;
}

/**
 * Возвраты товаров продавцу. Отчёт отдаётся не больше чем за 31 день за раз,
 * поэтому период режется на окна по 30 суток — с запасом на то, как метод
 * считает границы.
 *
 * Лимит суровый: один запрос в минуту. Год истории это дюжина окон и дюжина
 * минут, и всё это время проход занят — «С даты» стоит держать недалёкой.
 *
 * Артикула продавца в отчёте нет, только nmId и баркод, поэтому он берётся
 * из справочника карточек. Тот же справочник нужен остаткам ВБ, и он
 * кэшируется, так что лишнего обхода не будет.
 */
const ВОЗВРАТЫ_ВБ = '/api/v1/analytics/goods-return';
const ЛИСТ_ВОЗВРАТОВ_ВБ = 'FBS Возвраты';
const ОКНО_ВОЗВРАТОВ = 30;      // суток в одном запросе, предел метода — 31
const ОКОН_МАКСИМУМ = 24;       // два года: дальше прогон растянется на полчаса

const датаВБ = (д) => локальныйISO(д).slice(0, 10);

export async function wbВозвратыСтроки(задача) {
  const граница = границаИстории(задача || {});
  const по = new Date();
  const с = граница || new Date(по.getTime() - (ОКНО_ВОЗВРАТОВ - 1) * 24 * 3600 * 1000);
  if (с.getTime() > по.getTime()) {
    throw new Error(`«С даты» ${датаВБ(с)} позже сегодняшнего дня`);
  }

  const поБаркоду = await карточкиПоБаркоду();
  const собрано = [];
  const виденные = new Set();
  let окон = 0;

  for (let окноС = new Date(с); окноС.getTime() <= по.getTime();) {
    if (окон >= ОКОН_МАКСИМУМ) {
      throw new Error(`WB возвраты: период длиннее ${ОКОН_МАКСИМУМ} окон по `
        + `${ОКНО_ВОЗВРАТОВ} суток — поднимите «С даты»`);
    }
    const окноПо = new Date(Math.min(
      окноС.getTime() + (ОКНО_ВОЗВРАТОВ - 1) * 24 * 3600 * 1000, по.getTime()));
    окон += 1;

    // один запрос в минуту, и это предел на весь аккаунт
    await неЧаще('wb-возвраты', подставной(WB_АНАЛИТИКА) ? 0 : 61000);
    const ответ = await аналитика(
      `${ВОЗВРАТЫ_ВБ}?dateFrom=${датаВБ(окноС)}&dateTo=${датаВБ(окноПо)}`);
    const отчёт = ответ?.report ?? ответ?.data?.report;
    if (!Array.isArray(отчёт)) {
      // пустое тело или ответ чужой формы. Счесть это «возвратов нет» значило бы
      // записать успешный прогон там, где отчёта не было вовсе
      throw new Error(`WB возвраты: метод ответил без отчёта за `
        + `${датаВБ(окноС)} — ${датаВБ(окноПо)}`);
    }

    for (const в of отчёт) {
      const ключ = `${в.srid ?? ''}|${в.barcode ?? ''}`;
      if (виденные.has(ключ)) continue;
      виденные.add(ключ);
      собрано.push(в);
    }
    окноС = new Date(окноПо.getTime() + 24 * 3600 * 1000);
  }

  const отметка = вТаблицу(new Date());
  const строки = собрано.map((в) => {
    const баркод = String(в.barcode ?? '');
    const к = поБаркоду.get(баркод) || {};
    return [
      String(в.srid ?? ''), в.orderId ?? '-',
      текстом(баркод), текстом(к.артикул) || '-', в.nmId ?? '-',
      текстом(в.shkId), текстом(в.stickerId) || '-',
      в.subjectName || '-', в.brand || '-', текстом(в.techSize) || '-',
      в.reason || '-', в.returnType || '-', в.status || '-',
      Number(в.isStatusActive) ? 'да' : '-',
      вТаблицу(в.orderDt), вТаблицу(в.readyToReturnDt),
      вТаблицу(в.completedDt), вТаблицу(в.expiredDt),
      в.dstOfficeAddress || '-', в.dstOfficeId ?? '-',
      отметка,
    ];
  });

  const безКарточки = строки.filter((р) => р[3] === '-').length;
  return { строки, окон, с, безКарточки };
}

export async function wbВозвраты(задача) {

}

/**
 * Лента заказов — задача «WB заказы проверка». Отдельный метод на хосте
 * аналитики, со своим токеном, своим счётчиком лимита и своим расписанием:
 * с «Заказы ВБ FBO» и «Заказы ВБ FBS» он не связан, лист нужен для сверки.
 *
 * Три особенности, которые видны прямо в листе:
 *
 * 1. **Период считается по времени текущего статуса**, а не по дате оформления.
 *    Заказ, оформленный месяц назад и отменённый вчера, попадёт в сегодняшнее
 *    окно — это и нужно для сверки. Отсюда же следствие: по колонке «Дата
 *    заказа» период не сходится, она бывает раньше начала окна, и это не ошибка.
 * 2. **Глубже 31 суток метод не отдаёт ничего** — на более раннюю дату отвечает
 *    400. Окно обрезается до тридцати суток, и «С даты» старше месяца этому
 *    листу истории не добавит, сколько её ни правь.
 * 3. **Пагинация идёт `offset`-ом, но по снимку.** Со второго запроса передаётся
 *    `snapshotTime` из ПЕРВОГО ответа, а в первом его быть не должно — метод
 *    начал бы листать чужой, уже закрытый снимок.
 *
 * Пределы вынесены наружу, чтобы проба могла закрепить их соотношение с ёмкостью
 * листа: пятьдесят страниц по десять тысяч — это полмиллиона заказов при ёмкости
 * листа в пятьдесят тысяч, и каждая страница стоит минуты.
 */
const ЛЕНТА = '/api/analytics/v1/order-feed';
export const ПРЕДЕЛЫ_ЛЕНТЫ = { пачка: 10000, страниц: 10, глубинаСуток: 30 };

/**
 * Отпечаток страницы — по ОПОЗНАЮЩИМ полям, а не по состоянию: статус и его
 * время меняются на ходу, и застрявшая страница перестала бы узнаваться.
 *
 * Застрявший `offset` ловится именно отпечатком, а не признаком «нет новых
 * srid»: второе означало бы ещё и «srid повторяется», а это законно — и честная
 * выборка падала бы с жалобой на пагинацию.
 */
const отпечатокСтраницы = (пачка) => пачка
  .map((з) => `${з.srid ?? ''}|${з.nmId ?? ''}|${з.chrtId ?? ''}`).join(';');

export async function wbЛентаЗаказовСтроки(задача) {
  const { с, по, растянутоЗнаком } = период(задача);
  const предел = new Date(Date.now() - ПРЕДЕЛЫ_ЛЕНТЫ.глубинаСуток * 24 * 3600 * 1000);
  const окноС = с.getTime() < предел.getTime() ? предел : с;
  // Окно, растянутое простоем, пришлось обрезать — это дыра в истории, и назвать
  // её надо отдельно от штатного излишка: один текст на оба случая — это ложь,
  // а изменения статусов за простой уже не вернутся
  const пропуск = растянутоЗнаком && с.getTime() < предел.getTime() ? { с, по: предел } : null;

  const собрано = [];
  const виденные = new Set();
  const отпечатки = new Set();     // отпечатки ВСЕХ страниц: выборка ходит и по кругу A→B→A
  let снимок = '';
  let свёрнуто = 0;
  let страниц = 0;

  for (let offset = 0; ; offset += ПРЕДЕЛЫ_ЛЕНТЫ.пачка) {
    if (страниц >= ПРЕДЕЛЫ_ЛЕНТЫ.страниц) {
      throw new Error(`WB лента: прочитано ${собрано.length} заказов и упёрлись в предел `
        + `${ПРЕДЕЛЫ_ЛЕНТЫ.страниц} страниц — сузьте период`);
    }
    await неЧаще('wb-лента', подставной(WB_АНАЛИТИКА) ? 0 : 61000);

    const просьба = {
      dateFrom: сСмещением(окноС), dateTo: сСмещением(по),
      limit: ПРЕДЕЛЫ_ЛЕНТЫ.пачка, offset,
    };
    // в первом запросе снимка быть не должно
    if (снимок) просьба.snapshotTime = снимок;

    let ответ;
    try {
      ответ = await аналитика(ЛЕНТА, просьба);
    } catch (е) {
      // голое «HTTP 429» отправляет человека искать поломку в коде, а причина
      // бывает в категории токена
      if (/HTTP 429/.test(е.message)) {
        throw new Error(`${е.message} — лимит ленты: один запрос в минуту на аккаунт, `
          + 'а у токена категории «Базовый» — один запрос в три часа, и выборка '
          + 'длиннее одной страницы на нём невозможна в принципе');
      }
      throw е;
    }
    страниц += 1;

    const тело = ответ.data || ответ;
    const пачка = тело.orders;
    if (!Array.isArray(пачка)) {
      throw new Error(`WB лента: метод ответил без заказов на странице ${страниц}`);
    }
    // Данные ленты обновляются в реальном времени. Переснимет метод данные —
    // ответит 200 с НОВЫМ снимком, а мы продолжим листать по чужой нумерации
    const пришёл = String(тело.snapshotTime ?? '');
    if (!снимок) снимок = пришёл;
    else if (пришёл && пришёл !== снимок) {
      throw new Error(`WB лента: метод переснял данные (${снимок} → ${пришёл}) — `
        + 'дальше листалась бы чужая нумерация');
    }

    // Конец выборки подтверждает ПУСТАЯ страница, а не короткая: ни `has_next`,
    // ни общего числа метод не отдаёт, поэтому «страница короче предела» — это
    // догадка, и короткий ответ посреди обхода обрывал выгрузку молча. Лишний
    // запрос стоит минуты, потеря половины заказов — дороже
    if (!пачка.length) break;

    const отпечаток = отпечатокСтраницы(пачка);
    if (отпечатки.has(отпечаток)) {
      throw new Error(`WB лента: страница ${страниц} повторяет уже прочитанную — `
        + 'offset не двигает выборку');
    }
    отпечатки.add(отпечаток);

    for (const з of пачка) {
      const ключ = String(з.srid ?? '');
      if (ключ && виденные.has(ключ)) { свёрнуто += 1; continue; }
      if (ключ) виденные.add(ключ);
      собрано.push(з);
    }
  }

  // Справочник карточек обязателен. Прежде при его отказе в строку ставились
  // прочерки — «мягкая деградация». Но `сохранить()` пишет строку ЦЕЛИКОМ, то
  // есть затирала уже выгруженные артикулы, а вернуть их нельзя: заказ со
  // статусом старше тридцати суток лента больше не отдаёт
  let поРазмерам;
  try {
    поРазмерам = await карточкиПоРазмерам();
  } catch (е) {
    throw new Error(`WB лента: справочник карточек недоступен (${е.message}) — `
      + 'выгрузка остановлена, иначе прочерки затёрли бы артикулы в книге');
  }

  const отметка = вТаблицу(new Date());
  const строки = собрано.map((з) => {
    const к = поРазмерам.get(String(з.chrtId ?? '')) || {};
    const статус = String(з.status ?? '');
    return [
      // srid служит ключом строки, а он бывает девятнадцатизначным: без
      // принудительной пометки Таблицы потеряли бы хвост
      строгоТекстом(з.srid), вТаблицу(з.date), вТаблицу(з.updatedAt),
      текстом(к.артикул) || '-', текстом(к.баркод) || '', з.nmId || '-',
      число(з.quantity ?? 1), число(з.finishedPrice ?? з.totalPrice),
      статус || '-',
      // «Тип отмены» определён только при отмене. На проданном заказе причина
      // отмены — ложь, даже если метод её прислал
      статус === 'cancel' ? (з.cancelType || '-') : '-',
      з.warehouseName || '-', з.regionName || з.oblastOkrugName || '-', отметка,
    ];
  });

  const заметки = [];
  if (пропуск) {
    // дыру в истории надо называть, а не прятать за обычным текстом про период
    заметки.push(`ПРОПУСК ИСТОРИИ: ${вТаблицу(пропуск.с).slice(0, 16)} — `
      + `${вТаблицу(пропуск.по).slice(0, 16)}`);
  }
  // свёрнутые повторы считаем: усохшая в разы выборка иначе выглядит маленькой,
  // но полной
  if (свёрнуто) заметки.push(`свёрнуто повторов ${свёрнуто}`);
  return { строки, заметки, страниц, свёрнуто, пропуск };
}

export async function wbЛентаЗаказов(задача) {
  const { строки, заметки, страниц } = await wbЛентаЗаказовСтроки(задача);
  const итог = await сохранить('WB заказы проверка', строки);
  return `строк ${строки.length}, новых ${итог.новых}, изменено ${итог.изменено}`
       + `${итог.заметка}, страниц ${страниц}`
       + (заметки.length ? `, ${заметки.join(', ')}` : '');
}

/**
 * Остатки FBO собираются двумя методами, и иначе не выходит.
 *
 * `/v2/analytics/stock_on_warehouses` отдаёт весь список разом, по тысяче строк,
 * но из чисел в нём только три: доступное, зарезервированное и
 * `promised_amount` — «количество товара, указанное в подтверждённых будущих
 * поставках». Товара **в пути** среди них нет: сам метод в документации
 * предупреждает, что отличается от кабинетного «Отчёта по остаткам и товарам
 * в пути на склады Ozon». Раньше в колонку «В пути» уходил именно
 * `promised_amount` — то есть заявки на поставку, а не то, что едет на склад,
 * и у продавца без открытых заявок колонка стояла пустой.
 *
 * Товар в пути во всём Seller API есть ровно в одном поле:
 * `transit_stock_count` метода `/v1/analytics/stocks` — «количество единиц
 * товара в поставках в пути». Оттуда же приходит настоящий кластер: в ответе
 * v2 поля `cluster_name` нет вовсе, поэтому колонка «Кластер» до сих пор молча
 * повторяла «Склад».
 *
 * Платится это тем, что `/v1/analytics/stocks` — выборка по списку SKU, а не
 * дамп: `skus` обязателен и не длиннее сотни. Список берём из первого прохода,
 * он у нас уже на руках.
 */
const ПАЧКА_SKU = 100;
const ПАЧЕК_SKU_МАКСИМУМ = 500;

/** Ключ строки отчёта: товар на складе. */
const ключОстатка = (sku, склад) => `${sku}|${склад}`;

export async function ozonОстаткиFboСтроки() {
  const поКлючу = new Map();
  const скю = new Set();

  for (let n = 0, offset = 0; n < 200; n += 1) {
    // warehouse_type у метода — это не FBO/FBS, а фильтр по складам Ozon:
    // ALL, EXPRESS_DARK_STORE, NOT_EXPRESS_DARK_STORE. Значения 'FBO' в
    // перечислении нет вовсе, и метод молча брал ALL по умолчанию
    const ответ = await ozon('/v2/analytics/stock_on_warehouses',
      { limit: 1000, offset, warehouse_type: 'ALL' });
    const пачка = (ответ.result || ответ).rows;
    if (!Array.isArray(пачка)) {
      // пустой массив метод в ответе опускает, поэтому у продавца с числом строк,
      // кратным тысяче, следующая страница приходит как `{"result":{}}` — это
      // конец выдачи, а не сбой. На первой же странице списка нет — значит
      // форма ответа чужая, и молчать нельзя
      if (offset) break;
      throw new Error('Ozon остатки FBO: метод ответил без строк на первой странице');
    }
    for (const с of пачка) {
      const sku = String(с.sku ?? '');
      const склад = с.warehouse_name || '-';
      if (sku) скю.add(sku);
      поКлючу.set(ключОстатка(sku, склад), {
        артикул: с.item_code || '', sku, склад, кластер: '',
        доступно: число(с.free_to_sell_amount), вПути: 0,
        резерв: число(с.reserved_amount),
      });
    }
    if (пачка.length < 1000) break;
    offset += 1000;
    await сон(ПАУЗА_OZON);
  }

  // кластер и товар в пути — по тем же SKU, сотнями
  const список = [...скю];
  let безАналитики = 0;
  let сАналитикой = 0;
  for (let i = 0, п = 0; i < список.length; i += ПАЧКА_SKU, п += 1) {
    if (п >= ПАЧЕК_SKU_МАКСИМУМ) {
      throw new Error(`Ozon остатки FBO: ${список.length} SKU — больше `
        + `${ПАЧЕК_SKU_МАКСИМУМ * ПАЧКА_SKU}, аналитика не дочитана`);
    }
    if (п) await сон(ПАУЗА_OZON);
    const ответ = await ozon('/v1/analytics/stocks', { skus: список.slice(i, i + ПАЧКА_SKU) });
    const товары = (ответ.result || ответ).items;
    if (!Array.isArray(товары)) {
      // пустой ответ тут неотличим от «в пути ничего нет», а это и была та самая
      // молчаливая дыра: колонка пустая, прогон успешный
      throw new Error('Ozon остатки FBO: аналитика ответила без списка товаров');
    }
    for (const т of товары) {
      const sku = String(т.sku ?? '');
      const склад = т.warehouse_name || '-';
      const ключ = ключОстатка(sku, склад);
      const было = поКлючу.get(ключ);
      сАналитикой += 1;
      if (было) {
        было.кластер = т.cluster_name || '';
        было.вПути = число(т.transit_stock_count);
        continue;
      }
      // товар, которого на складе ещё нет, а поставка уже едет: в дампе v2 такой
      // строки нет, и без этой ветки «в пути» по нему не показалось бы вовсе
      поКлючу.set(ключ, {
        артикул: т.offer_id || '', sku, склад, кластер: т.cluster_name || '',
        доступно: число(т.available_stock_count), вПути: число(т.transit_stock_count),
        резерв: 0,
      });
    }
  }

  // Пустой список в одной пачке — обычное дело: по этим SKU ничего не едет.
  // Пустой во всех неотличим от «в пути ничего нет», а последствие у него другое:
  // «Кластер» уйдёт прочерком, «В пути» нулём по всему снимку, и это та самая
  // молчаливая дыра, только теперь с успешным прогоном. Дамп отдал строки —
  // значит аналитика обязана знать хоть один их SKU: это один и тот же отчёт FBO
  if (список.length && !сАналитикой) {
    throw new Error(`Ozon остатки FBO: аналитика не знает ни одного из ${список.length} SKU `
      + '— кластер и товар в пути остались бы пустыми при успешном прогоне');
  }

  const отметка = вТаблицу(new Date());
  const строки = [];
  for (const о of поКлючу.values()) {
    if (!о.доступно && !о.вПути && !о.резерв) continue;
    if (!о.кластер) безАналитики += 1;
    строки.push([текстом(о.артикул) || '-', о.sku, о.склад, о.кластер || '-',
      о.доступно, о.вПути, о.резерв, отметка]);
  }
  const вПути = строки.reduce((с, р) => с + р[5], 0);
  return { строки, вПути, безАналитики };
}

export async function ozonОстаткиFbo() {
  const { строки, вПути, безАналитики } = await ozonОстаткиFboСтроки();
  await сохранить('FBO остатки ОЗОН', строки);
  return `строк ${строки.length}, в пути ${вПути}`
       + (безАналитики ? `, без кластера ${безАналитики}` : '');
}

export async function ozonОстаткиFbs() {
  const строки = [];
  const отметка = вТаблицу(new Date());
  let cursor = '';
  for (let n = 0; n < 300; n += 1) {
    const ответ = await ozon('/v4/product/info/stocks',
      { cursor, limit: 1000, filter: { visibility: 'ALL' } });
    const тело = ответ.result || ответ;
    const товары = тело.items || [];
    for (const т of товары) {
      for (const с of т.stocks || []) {
        if (с.type && с.type !== 'fbs') continue;
        строки.push([текстом(т.offer_id) || '-', String(т.product_id ?? ''), 'Склад продавца',
          число(с.present), число(с.reserved), отметка]);
      }
    }
    cursor = тело.cursor || '';
    if (!cursor || товары.length < 1000) break;
    await сон(300);
  }
  await сохранить('FBS остатки ОЗОН', строки);
  return `строк ${строки.length}`;
}

/**
 * Списки отправлений отдают не больше сотни за раз: на 1000 метод отвечает
 * 400 «value must be inside range (0, 100]».
 *
 * Листать их `offset`-ом нельзя. У методов нового поколения такого поля нет,
 * а неизвестные поля метод молча выбрасывает — поэтому запрос выглядел рабочим,
 * отвечал 200 и на любой странице отдавал одну и ту же первую сотню.
 *
 * Обход идёт лестницей: следующая ступень включается сама, когда предыдущая
 * перестаёт двигать выборку.
 *   1. курсор метода;
 *   2. рез окна по дате последнего прочитанного отправления;
 *   3. делёж окна пополам.
 * Нижние две ступени опираются только на фильтр по датам, который заведомо
 * работает — им всё равно, чем именно метод листается и листается ли вообще.
 */
const ПАЧКА_ОТПРАВЛЕНИЙ = 100;
const ЗАПРОСОВ_МАКСИМУМ = 500;
// период длиннее года метод не берёт, а первичная заливка бывает и глубже.
// Режем с запасом: лишнее окно стоит один запрос
const ОКНО_ОТПРАВЛЕНИЙ = 180 * 24 * 3600 * 1000;

/** Одна страница. Поля запроса — только те, что у метода есть. */
async function страницаОтправлений(метод, с, по, курсор) {
  const тело = {
    limit: ПАЧКА_ОТПРАВЛЕНИЙ,
    filter: { since: iso(с), to: iso(по) },
    with: { analytics_data: true, financial_data: true },
    sort_dir: 'ASC',
  };
  if (курсор) тело.cursor = курсор;

  await сон(ПАУЗА_OZON);
  const ответ = await ozon(метод, тело);
  const т = ответ.result || ответ;
  const список = Array.isArray(т) ? т : (т.postings || null);
  if (!Array.isArray(список)) {
    // пустое тело, 204 или ответ чужой формы. Раньше это молча считалось
    // концом окна, то есть потеря данных выглядела как успешный прогон
    throw new Error(`метод ответил без списка отправлений на окне с ${вТаблицу(с)}`);
  }
  return {
    список,
    курсор: Array.isArray(т) ? '' : String(т.cursor || т.next_cursor || т.last_id || ''),
    естьЕщё: Array.isArray(т) ? null : т.has_next,
  };
}

const датаОтправления = (о) => new Date(о.created_at || о.in_process_at);

/**
 * Где резать окно. Годится любая точка строго внутри него, поэтому берём дату
 * последнего прочитанного отправления, а если она не внутри — середину.
 *
 * Двигать НАЧАЛО окна по дате страницы нельзя: это молчаливая ставка на то, что
 * метод сортирует ровно по тому полю, по которому считает период. Разойдись они —
 * начало перескочит через хвост сортировки, и отправления пропадут без единого
 * слова. Рез конца такой ставки не делает: две половины покрывают окно целиком
 * при любом месте реза.
 */
function точкаРеза(список, с, по) {
  const даты = список.map(датаОтправления)
    .filter((д) => !Number.isNaN(д.getTime())).map((д) => д.getTime());
  const макс = даты.length ? Math.max(...даты) : null;
  if (макс !== null && макс > с.getTime() && макс < по.getTime()) {
    return { дата: new Date(макс), способ: 'дата' };
  }
  const середина = Math.floor((с.getTime() + по.getTime()) / 2);
  if (середина > с.getTime() && середина < по.getTime()) {
    return { дата: new Date(середина), способ: 'делёж' };
  }
  return null;
}

async function обходОкна(метод, с, по, ст) {
  let курсор = '';
  let последняя = null;

  for (;;) {
    if (ст.запросов >= ЗАПРОСОВ_МАКСИМУМ) {
      throw new Error(`прочитано ${ст.собрано.length} отправлений и упёрлись в предел `
        + `${ЗАПРОСОВ_МАКСИМУМ} запросов — сузьте период`);
    }
    ст.запросов += 1;
    const стр = await страницаОтправлений(метод, с, по, курсор);
    последняя = стр.список;

    let свежих = 0;
    for (const о of стр.список) {
      const ключ = String(о.posting_number || '');
      if (ключ && ст.виденные.has(ключ)) continue;
      if (ключ) ст.виденные.add(ключ);
      ст.собрано.push(о);
      свежих += 1;
    }

    // короткая страница — конец окна, но только если метод не сказал прямо, что
    // есть ещё: недолить страницу он вправе, и вера в её длину обрезала бы
    // выборку молча. Сам `has_next: false` конец окна закрывает при любой длине
    if (стр.список.length < ПАЧКА_ОТПРАВЛЕНИЙ && стр.естьЕщё !== true) return;
    if (стр.естьЕщё === false) return;
    if (стр.курсор && стр.курсор !== курсор && свежих) {
      курсор = стр.курсор;
      ст.способы.add('курсор');
      continue;
    }
    break;   // курсор не двигает выборку — дальше идём по датам
  }

  // вся полная страница пришла одним мгновением: никакой рез её не разъединит,
  // а курсора нет. Ловим сразу, иначе обход сорок раз поделит окно пополам
  // и упрётся в предел запросов, сообщив совсем не о том
  const мгновения = new Set(последняя.map((о) => датаОтправления(о).getTime()));
  if (мгновения.size === 1) {
    throw new Error(`окно на ${вТаблицу(последняя[0] && датаОтправления(последняя[0]))} `
      + `уже не делится, а ${ПАЧКА_ОТПРАВЛЕНИЙ} отправлений в нём не кончаются`);
  }

  const рез = точкаРеза(последняя, с, по);
  if (!рез) {
    throw new Error(`окно на ${вТаблицу(с)} уже не делится, `
      + `а ${ПАЧКА_ОТПРАВЛЕНИЙ} отправлений в нём не кончаются`);
  }
  ст.способы.add(рез.способ);
  if (!ст.способы.has('курсор')) console.warn(`[${метод}] курсор не двигает выборку, дальше идём по дате`);

  await обходОкна(метод, с, рез.дата, ст);
  // правая половина начинается миллисекундой РАНЬШЕ реза: понимание границ
  // у метода неизвестно, и пойми он строго обе — отправление ровно на шве
  // выпало бы из обеих половин. Перекрытие снимет указатель прочитанных
  await обходОкна(метод, new Date(рез.дата.getTime() - 1), по, ст);
}

export async function ozonОтправления(метод, задача, кластер) {
  const { с, по } = период(задача);
  const ст = { виденные: new Set(), собрано: [], запросов: 0, способы: new Set() };
  // длинный период сразу режем на окна по пределу метода, не дожидаясь от него 400
  for (let край = с; ;) {
    const конец = new Date(Math.min(край.getTime() + ОКНО_ОТПРАВЛЕНИЙ, по.getTime()));
    await обходОкна(метод, край, конец, ст);
    if (конец.getTime() >= по.getTime()) break;
    // шов перекрываем на миллисекунду — по той же причине, что и при резе окна
    край = new Date(конец.getTime() - 1);
  }

  const отметка = вТаблицу(new Date());
  const строки = ст.собрано.flatMap((о) => {
    const а = о.analytics_data || {};
    const ф = о.financial_data || {};
    return (о.products || []).map((т) => [
      о.posting_number, о.order_number || '-',
      вТаблицу(датаОтправления(о)),
      текстом(т.offer_id) || '-', String(т.sku), число(т.quantity), число(т.price),
      о.status || '-',
      // кластер приходит только в financial_data, а склад у FBS зовётся warehouse,
      // а не warehouse_name: перебираем оба, чтобы не зависеть от схемы метода
      кластер
        ? (ф.cluster_to || а.cluster_to || а.warehouse_name || а.warehouse || '-')
        : (а.warehouse_name || а.warehouse || '-'),
      а.region || '-', отметка,
    ]);
  });

  const порядок = ['курсор', 'дата', 'делёж'].filter((с_) => ст.способы.has(с_));
  return {
    строки,
    сводка: `отправлений ${ст.собрано.length}, запросов ${ст.запросов}, `
          + `листали: ${порядок.join(' + ') || 'одна страница'}`,
  };
}

async function ozonЗаказы(метод, лист, задача, кластер) {
  const { строки, сводка } = await ozonОтправления(метод, задача, кластер);
  const итог = await сохранить(лист, строки);
  return `строк ${строки.length}, новых ${итог.новых}, изменено ${итог.изменено}`
       + `${итог.заметка}, ${сводка}`;
}

/**
 * Возвраты FBO и FBS приходят одним методом, поэтому лист под них один, а схема
 * конкретного возврата пишется в колонку «Схема».
 *
 * Период берётся из «С даты» как есть: весь отрезок от неё до сегодня, каждый
 * прогон заново. Глубина и водяной знак тут не участвуют — метод принимает
 * только один фильтр-дату за запрос, и дробить отрезок нечем. Пустая «С даты»
 * значит «все возвраты, какие есть».
 *
 * Фильтруем по дате возврата: это та же дата, что уходит в колонку
 * «Дата возврата», поэтому период в книге совпадает с тем, что видно в листе.
 *
 * Листается `last_id` — идентификатор последнего возврата на странице.
 */
const ВОЗВРАТЫ = '/v1/returns/list';
const ЛИСТ_ВОЗВРАТОВ = 'Возвраты OZON';
const ПАЧКА_ВОЗВРАТОВ = 500;      // предел метода
const СТРАНИЦ_ВОЗВРАТОВ = 200;    // сто тысяч возвратов

/** Суммы приходят объектом {currency_code, price}, причём price — строкой. */
const деньги = (з) => число(з && з.price);

/** Сбор и раскладка по колонкам, без записи в книгу: так его проверяет проба. */
export async function ozonВозвратыСтроки(задача) {
  const с = границаИстории(задача || {});
  const фильтр = с
    ? { logistic_return_date: { time_from: iso(с), time_to: iso(new Date()) } }
    : null;

  const собрано = [];
  const виденные = new Set();
  let last = 0;
  let дочитано = false;

  for (let n = 0; n < СТРАНИЦ_ВОЗВРАТОВ && !дочитано; n += 1) {
    await сон(ПАУЗА_OZON);
    const тело = { limit: ПАЧКА_ВОЗВРАТОВ, last_id: last };
    if (фильтр) тело.filter = фильтр;
    const ответ = await ozon(ВОЗВРАТЫ, тело);
    const ответТело = ответ.result || ответ;
    // пустое тело, 204 или ответ чужой формы. Молча счесть это концом списка
    // значило бы записать «возвратов 0» как успешный прогон
    if (!Array.isArray(ответТело.returns)) {
      throw new Error(`возвраты: метод ответил без списка на странице ${n + 1}`);
    }
    const пачка = ответТело.returns;

    for (const в of пачка) {
      const ключ = String(в.id ?? '');
      if (ключ && виденные.has(ключ)) continue;
      if (ключ) виденные.add(ключ);
      собрано.push(в);
    }

    if (!пачка.length || ответТело.has_next === false) { дочитано = true; break; }
    if (ответТело.has_next === undefined && пачка.length < ПАЧКА_ВОЗВРАТОВ) {
      дочитано = true; break;
    }

    const хвост = Number(пачка[пачка.length - 1].id);
    // та же беда, что была у отправлений: курсор, который не двигает выборку,
    // выглядит как бесконечный список одних и тех же записей
    if (!Number.isSafeInteger(хвост) || хвост === last) {
      throw new Error(`возвраты: last_id не двигает выборку, прочитано ${собрано.length}`);
    }
    last = хвост;
  }
  if (!дочитано) {
    throw new Error(`возвраты: прочитано ${собрано.length} и упёрлись `
      + `в предел ${СТРАНИЦ_ВОЗВРАТОВ} страниц`);
  }

  const отметка = вТаблицу(new Date());
  const строки = собрано.map((в) => {
    const т = в.product || {};
    const вид = в.visual || {};
    const лог = в.logistic || {};
    return [
      String(в.id ?? ''), в.posting_number || '-', в.order_number || '-',
      в.schema || '-',
      текстом(т.offer_id) || '-', String(т.sku ?? ''), текстом(т.name) || '-',
      число(т.quantity), деньги(т.price),
      в.return_reason_name || '-', в.type || '-',
      вид.status?.display_name || '-', вТаблицу(вид.change_moment),
```

```js
function изЯМ(строка) {
  if (!строка) return null;
  const м = String(строка).match(/^(\d{2})-(\d{2})-(\d{4})(?:\s+(\d{2}):(\d{2}):(\d{2}))?$/);

 * Остатки приходят разложенными по типам, и типы **пересекаются**:
 *
 *   `AVAILABLE` («Доступный к заказу») — товар, доступный для продажи;
 *   `FREEZE` — товар, зарезервированный под заказы;
 *   `FIT` («Годный») — товар, доступный для продажи ИЛИ уже зарезервированный,
 *   то есть `AVAILABLE` + `FREEZE`.
 *
 * Складывать `AVAILABLE` с `FIT`, как если бы это были два имени одного и того
 * же, нельзя — свободный остаток посчитается дважды. Ровно на этом лист показывал
 * двойные остатки и по FBO, и по FBS: при десяти штуках без резерва и в «Доступно»,
 * и в «Всего» стояло двадцать.
 *
 * Остальные типы — `DEFECT`, `EXPIRED`, `QUARANTINE`, `UTILIZATION` — лежат рядом
 * с годным, а не внутри него, поэтому в «Всего» идут как есть.
 */
const ВНУТРИ_ГОДНОГО = new Set(['AVAILABLE', 'FREEZE']);

function остатокТовара(товар) {
  const поТипу = new Map();
  for (const о of товар.stocks || []) {
    поТипу.set(о.type, (поТипу.get(о.type) || 0) + число(о.count));
  }
  const заморожено = поТипу.get('FREEZE') || 0;
  // не дал «годный» — собираем его из частей; не дал «доступный» — вычитаем резерв
  const годный = поТипу.has('FIT')
    ? поТипу.get('FIT')
    : (поТипу.get('AVAILABLE') || 0) + заморожено;
  const доступно = поТипу.has('AVAILABLE')
    ? поТипу.get('AVAILABLE')
    : Math.max(0, годный - заморожено);
  let прочее = 0;
  for (const [тип, к] of поТипу) {
    if (тип !== 'FIT' && !ВНУТРИ_ГОДНОГО.has(тип)) прочее += к;
  }
  return { доступно, заморожено, брак: поТипу.get('DEFECT') || 0, всего: годный + прочее };
}
```

```js
function поддельныйВБ({ страницы = [], снимки = null, безКарточек = false, лимит429 = false } = {}) {
  const запросы = [];
  const сервер = http.createServer((req, res) => {
    let сырое = '';
    req.on('data', (ч) => { сырое += ч; });
    req.on('end', () => {
      const u = new URL(req.url, 'http://x');

        res.writeHead(429, { 'content-type': 'application/json' });
        return res.end(JSON.stringify({ title: 'too many requests' }));
      }
      const тело = JSON.parse(сырое || '{}');
      запросы.push(тело);
      const н = запросы.length - 1;
      res.writeHead(200, { 'content-type': 'application/json' });
```

```js
/**
 * Поддельный Ozon на два метода.
 * `безСтрок` / `безТоваров` — ответ нужной формы, но без списка.
 */
function поддельныйOzon({ строкиДампа = [], товарыАналитики = [], безСтрок = false,
  безТоваров = false } = {}) {
  const запросы = [];
  const сервер = http.createServer((req, res) => {
    let сырое = '';
    req.on('data', (ч) => { сырое += ч; });
    req.on('end', () => {
      const тело = JSON.parse(сырое || '{}');
      запросы.push({ путь: req.url, тело });
      const отказ = (т) => {
        res.writeHead(400, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ code: 3, message: т }));
      };
      const ответить = (о) => {
        res.writeHead(200, { 'content-type': 'application/json' });
        res.end(JSON.stringify(о));
      };

      if (req.url === '/v2/analytics/stock_on_warehouses') {
        for (const к of Object.keys(тело)) {
          if (!ПОЛЯ_ДАМПА.has(к)) return отказ(`unknown field "${к}"`);
        }
        if (тело.warehouse_type !== undefined && !ТИПЫ_СКЛАДА.has(тело.warehouse_type)) {
          return отказ(`invalid warehouse_type "${тело.warehouse_type}"`);
        }
        if (безСтрок) return ответить({ result: {} });
        const от = тело.offset || 0;
        const кусок = строкиДампа.slice(от, от + тело.limit);
        // пустой массив настоящий метод в ответе опускает (protojson), и у
        // продавца с числом строк, кратным тысяче, так приходит последняя страница
        return ответить({ result: кусок.length ? { rows: кусок } : {} });
      }

      if (req.url === '/v1/analytics/stocks') {
        for (const к of Object.keys(тело)) {
          if (!ПОЛЯ_АНАЛИТИКИ.has(к)) return отказ(`unknown field "${к}"`);
        }
        // skus обязателен и не длиннее сотни — так в схеме метода
        if (!Array.isArray(тело.skus) || !тело.skus.length) return отказ('skus is required');
        if (тело.skus.length > 100) return отказ('skus: maximum 100');
        if (безТоваров) return ответить({});
        const нужные = new Set(тело.skus.map(String));
        return ответить({ items: товарыАналитики.filter((т) => нужные.has(String(т.sku))) });
      }
      return отказ(`unknown path ${req.url}`);
    });
  });
  return { сервер, запросы };
}

async function собрать(настройки) {
  const { сервер, запросы } = поддельныйOzon(настройки);
  await new Promise((r) => сервер.listen(0, '127.0.0.1', r));
  process.env.OZON_API_URL = `http://127.0.0.1:${сервер.address().port}`;
  const { ozonОстаткиFboСтроки } = await import(`../src/wb-ozon.js?${Math.random()}`);
  try {
    return { ...(await ozonОстаткиFboСтроки()), запросы };
  } finally {
    сервер.close();
  }
}
```

```js
 * Типы остатков у метода **пересекаются** — `FIT` («Годный») это `AVAILABLE`

 * `страницы` — по элементу на страницу, каждый со своим списком складов.
 * `токенНеДвигается` — метод отдаёт один и тот же `page_token` и ту же страницу:
 * так проверяется, что повторная страница не задваивает товар.
 */
function поддельныйЯМ({ страницы, токенНеДвигается = false, безСкладов = false }) {
  const запросы = [];
  const сервер = http.createServer((req, res) => {
    let сырое = '';
    req.on('data', (ч) => { сырое += ч; });
    req.on('end', () => {
      const u = new URL(req.url, 'http://x');
      запросы.push({ токен: u.searchParams.get('page_token'), лимит: u.searchParams.get('limit') });
      res.writeHead(200, { 'content-type': 'application/json' });
      if (безСкладов) return res.end(JSON.stringify({ result: { paging: {} } }));

      const н = токенНеДвигается ? 0 : запросы.length - 1;
      const склады = страницы[Math.min(н, страницы.length - 1)] || [];
      const ещё = токенНеДвигается || н < страницы.length - 1;
```

```js
  недолив = 0, пределПериода = 0,
}) {
  const запросы = [];
  const сервер = http.createServer((req, res) => {
    let сырое = '';
    req.on('data', (ч) => { сырое += ч; });
    req.on('end', () => {
      const тело = JSON.parse(сырое || '{}');
      запросы.push(тело);
      const отказ = (текст) => {
        res.writeHead(400, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ code: 3, message: текст }));
      };

      for (const к of Object.keys(тело)) {
        if (!ПОЛЯ.has(к)) return отказ(`unknown field "${к}"`);
      }
      for (const к of Object.keys(тело.filter || {})) {
        if (!ПОЛЯ_ФИЛЬТРА.has(к)) return отказ(`unknown filter field "${к}"`);
      }
      if (!(тело.limit > 0 && тело.limit <= 100)) {
        return отказ('invalid Limit: value must be inside range (0, 100]');

      const с = new Date(тело.filter.since).getTime();
      const по = new Date(тело.filter.to).getTime();
      // живой метод не берёт период длиннее года
      if (пределПериода && по - с > пределПериода) return отказ('PERIOD_IS_TOO_LONG');

      // так проверяется, что обход не двигает начало окна по дате страницы
      const подходят = отправления
        .filter((о) => { const д = new Date(о.created_at).getTime(); return д >= с && д <= по; })
        .sort((а, б) => String(а[полеСортировки]).localeCompare(String(б[полеСортировки])));
```

```js
#!/usr/bin/env node
/**
 * Документация против кода — по содержимому, а не по номерам строк.
 *
 * Номера строк оказались хуже, чем ничего: номер устаревает от любой вставки, а
 * проверка «строка существует и не пуста» съехавшую ссылку пропускает — документ
 * продолжает выглядеть точным, указывая не туда. Попались дважды, во второй раз
 * соврали сразу тридцать две ссылки. Поэтому номеров в документации нет вовсе,
 * а сверяется то, что названо: файлы, адреса методов, имена констант,
 * переменные окружения, диапазоны настроек, имена листов.
 *
 * Сверка идёт ТОЛЬКО с `src/`. Пробы держат поддельные серверы с теми же
 * адресами, служебные скрипты — пояснения с именами констант, а свод и вовсе
 * содержит весь код целиком: искать доказательство там значит доказывать
 * документацией документацию. Один раз эта проверка нашла имя константы в
 * собственном комментарии и промолчала.
 */
import { readFile, readdir } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const КОРЕНЬ = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const ИСХОДНИКИ = 'src';

/** Что в документации не ссылка, а просто слово из прозы. */
const НЕ_ССЫЛКИ = new Set([
  'FBO', 'FBS', 'FBY', 'FBP', 'API', 'JSON', 'HTTP', 'HTTPS', 'ESM', 'UTC', 'ISO',
  'SKU', 'URL', 'RPC', 'REST', 'WB', 'ВБ', 'OZON', 'ОЗОН', 'ДД', 'ММ', 'ГГГГ',
  'НЕ', 'ПАГИНАЦИЯ', 'ЛИСТАЕТ', 'ОШИБКА', 'ПРОПУСК', 'ИСТОРИИ', 'ФИЛЬТР',
  'ПРИМЕНЁН', 'ВСЕ', 'ИЗ', 'И', 'ИЛИ', 'НИ', 'РАНЬШЕ', 'ЦЕЛИКОМ',
]);

async function кодИсходников() {
  const имена = (await readdir(path.join(КОРЕНЬ, ИСХОДНИКИ))).filter((и) => и.endsWith('.js'));
  const куски = await Promise.all(имена.map((и) =>
    readFile(path.join(КОРЕНЬ, ИСХОДНИКИ, и), 'utf8')));
  return куски.join('\n');
}

async function есть(относительный) {
  try {
    await readFile(path.join(КОРЕНЬ, относительный), 'utf8');
    return true;
  } catch {
    return false;
  }
}

const код = await кодИсходников();
const документы = (await readdir(path.join(КОРЕНЬ, 'docs')))
  .filter((и) => и.endsWith('.md') && и !== 'ves-kod.md');
документы.push('../README.md');

const беды = [];
let сверено = 0;
const пожаловаться = (док, что, чего) => беды.push(`${док}: ${что} — ${чего}`);

for (const док of документы) {
  const текст = await readFile(path.join(КОРЕНЬ, 'docs', док), 'utf8');
  const имя = path.basename(док);

  // 1. номеров строк не должно быть вовсе
  for (const м of текст.matchAll(/`[A-Za-zа-яё\-]+\.m?js:\d+`/g)) {
    пожаловаться(имя, м[0], 'ссылка по номеру строки: номера устаревают молча');
  }

  // 2. названные файлы существуют
  for (const м of текст.matchAll(/`((?:src|scripts|проба|docs)\/[^`]+?)`/g)) {
    сверено += 1;
    if (!(await есть(м[1]))) пожаловаться(имя, м[1], 'такого файла нет');
  }

  // 3. адреса методов встречаются в src/
  for (const м of текст.matchAll(/`((?:GET|POST)\s+)?(\/(?:api|v\d|rpc|campaigns)[\w/{}.-]*)/g)) {
    const адрес = м[2].replace(/\{[^}]*\}/g, '');
    const куски = адрес.split(/\{|\}/)[0].split('/').filter(Boolean);
    const искать = куски.length > 1 ? куски.slice(-2).join('/') : куски[0];
    if (!искать || искать.length < 4) continue;
    сверено += 1;
    if (!код.includes(искать)) пожаловаться(имя, м[2], `в ${ИСХОДНИКИ}/ такого адреса нет`);
  }

  // 4. имена констант и функций встречаются в src/
  for (const м of текст.matchAll(/`([А-ЯЁ][А-ЯЁ_]{2,}|[а-яёA-Za-z][А-Яа-яёЁA-Za-z_]*\(\))`/g)) {
    const слово = м[1].replace(/\(\)$/, '');
    if (НЕ_ССЫЛКИ.has(слово)) continue;
    сверено += 1;
    if (!код.includes(слово)) пожаловаться(имя, м[1], `в ${ИСХОДНИКИ}/ такого имени нет`);
  }

  // 5. переменные окружения
  for (const м of текст.matchAll(/`([A-Z][A-Z0-9_]{3,})`/g)) {
    if (НЕ_ССЫЛКИ.has(м[1])) continue;
    сверено += 1;
    if (!код.includes(м[1])) пожаловаться(имя, м[1], `в ${ИСХОДНИКИ}/ такой переменной нет`);
  }

  // 6. диапазоны листа настроек сходятся с НАСТРОЙКИ
  const настройки = await readFile(path.join(КОРЕНЬ, ИСХОДНИКИ, 'schemas.js'), 'utf8');
  const числоИз = (имя_) => Number((настройки.match(new RegExp(`${имя_}:\\s*(\\d+)`)) || [])[1]);
  const диапазоны = {
    'A3:B12': ['параметрыОт', 'параметрыДо'],
    'A15:G61': ['задачиОт', 'задачиДо'],
  };
  for (const м of текст.matchAll(/`?\bA(\d+):([A-Z])(\d+)\b`?/g)) {
    const ключ = `A${м[1]}:${м[2]}${м[3]}`;
    const пара = диапазоны[ключ];
    сверено += 1;
    if (!пара) {
      // диапазон, которого нет среди известных: либо опечатка, либо настройки съехали
      const известные = Object.keys(диапазоны).join(', ');
      пожаловаться(имя, ключ, `неизвестный диапазон (знаем: ${известные})`);
      continue;
    }
    if (числоИз(пара[0]) !== +м[1] || числоИз(пара[1]) !== +м[3]) {
      пожаловаться(имя, ключ, `в schemas.js это ${числоИз(пара[0])}–${числоИз(пара[1])}`);
    }
  }

  // 7. имена листов в кавычках-ёлочках. Имя бывает перенесено по строкам —
  // в прозе это законно, поэтому пробелы сводим к одному. Шаблоны вида
  // «Заказы *» — не имена, а обобщение, и проверять их нечем
  for (const м of текст.matchAll(/«((?:FBO|FBS|Заказы|Возвраты|WB)[^»]{2,60})»/g)) {
    const лист = м[1].replace(/\s+/g, ' ').trim();
    if (лист.includes('*')) continue;
    сверено += 1;
    if (!код.includes(лист)) пожаловаться(имя, `«${лист}»`, 'такого листа нет в схемах');
  }
}

for (const б of беды) console.log(б);
console.log(`\nсверено ${сверено} упоминаний, расхождений ${беды.length}`);
if (беды.length) process.exitCode = 1;
```
