import { описаниеПериода } from './http.js';
import * as wbOzon from './wb-ozon.js';

const ОБРАБОТЧИКИ = {

  'FBS остатки ВБ': wbOzon.wbОстаткиFbs,

  // у М.Видео публичного API продавца нет: эти листы заполняются вручную
};

async function выполнить(задача) {
  const обработчик = ОБРАБОТЧИКИ[задача.имя];
  if (!обработчик) {
    await отметитьЗапуск(задача, 'нет коннектора, данные заливаются вручную');
    return;
  }
  const старт = Date.now();
  try {
    const результат = await обработчик(задача);
    const сек = ((Date.now() - старт) / 1000).toFixed(1);
    // период показываем только у задач, которые его используют: у остатков он смысла не несёт
    const подпись = задача.имя.startsWith('Заказы') ? описаниеПериода(задача) : '';
    const хвост = подпись ? `, ${подпись}` : '';
    await отметитьЗапуск(задача, `${результат}, ${сек} c${хвост}`);
    console.log(`[ok] ${задача.имя}: ${результат} (${сек} c)`);
  } catch (е) {
    await отметитьЗапуск(задача, `ОШИБКА: ${е.message}`.slice(0, 400));
    console.error(`[fail] ${задача.имя}: ${е.message}`);
  }
}

/**
 * Проходы нельзя пускать внахлёст. Задача может идти дольше минуты, а таймер
 * тикает всё равно: тогда «Последний запуск» ещё не записан, `пора()` снова
 * говорит «да», и та же задача уходит в работу вторым и третьим экземпляром.
 */
let идёт = false;

async function проход(принудительно = null) {
  const список = await задачи();
  const кЗапуску = принудительно
    ? список.filter((з) => з.имя === принудительно)
    : список.filter(пора);
  if (принудительно && !кЗапуску.length) {
    console.error(`задача «${принудительно}» не найдена в блоке задач на листе настроек`);
    process.exitCode = 1;
    return;
  }
  if (!кЗапуску.length) return;
  console.log(`к запуску: ${кЗапуску.map((з) => з.имя).join(', ')}`);
  for (const задача of кЗапуску) {
    await выполнить(задача);
  }
}

const аргументы = process.argv.slice(2);
const одинРаз = аргументы.includes('--once');
const позиция = аргументы.indexOf('--task');
const однаЗадача = позиция >= 0 ? аргументы[позиция + 1] : null;

/** Тик расписания: пропускаем, если прошлый проход ещё не закончился. */
function тик() {
  if (идёт) {
    console.log('прошлый проход ещё идёт, тик пропущен');
    return;
  }
  идёт = true;
  // проход роняет процесс только в разовых режимах: воркер должен пережить
  // временную недоступность Таблиц и попробовать снова через минуту
  проход()
    .catch((е) => console.error('сбой прохода:', е.message))
    .finally(() => { идёт = false; });
}

if (однаЗадача) {
  await проход(однаЗадача);
} else if (одинРаз) {
  await проход();
} else {
  console.log('сервис запущен, проверка расписания раз в минуту');
  тик();
  setInterval(тик, 60000);
}

import { НАСТРОЙКИ, ЗАДАЧИ } from './schemas.js';
import { читать, писать } from './sheets.js';
import { вТаблицу, изТаблицы } from './http.js';

const ЛИСТ = НАСТРОЙКИ.лист;

/** Ключи берём из окружения — на листе их держать не обязательно. */
export const доступ = (имя) => process.env[имя] || '';

export function требовать(...имена) {
  const нет = имена.filter((и) => !доступ(и));
  if (нет.length) throw new Error(`не заполнено: ${нет.join(', ')}`);
  return имена.map(доступ);
}

export async function параметры() {
  const строки = await читать(
    `'${ЛИСТ}'!A${НАСТРОЙКИ.параметрыОт}:B${НАСТРОЙКИ.параметрыДо}`,
  );
  const карта = {};
  for (const [имя, знач] of строки) {
    if (имя) карта[String(имя).trim()] = знач;
  }
  return карта;
}

/**
 * Строка блока настроек → задача. Вынесено отдельно и без обращений к книге:
 * разбор колонок — самое хрупкое место, и проверять его надо без сети.
 */
export function задачаИзСтроки(р, строка) {
  // в блоке возвратов обе колонки, D и E, озаглавлены «С даты», и неизвестно,
  // в какую вписана дата. Принимаем из любой: настоящая дата в колонке глубины
  // глубиной быть не может, а глубина датой не притворится — год не тот
  // берём ту колонку, которая разбирается в настоящую дату, а не первую непустую:
  // пробел, ноль или оставшаяся подпись в E иначе отбросили бы дату из D.
  // Если ни одна не разбирается, отдаём E как есть — пусть границаИстории
  // скажет понятное «не разобрана «С даты»», а не тихо загрузит весь список
  const датаВГлубине = датойПохоже(р[3]);
  const сдаты = датойПохоже(р[4]) ? р[4] : (датаВГлубине ? р[3] : (р[4] ?? ''));
  const отметка = изТаблицы(р[5]);
  const результат = String(р[6] ?? '').trim();
  return {
    имя: String(р[0] ?? '').trim(),
    строка,
    включена: Number(р[1]) === 1,
    каждые: Number(р[2]) || 30,
    глубина: датаВГлубине ? 0 : глубинаИз(р[3]),
    // значение ячейки держим сырым: дата из книги приходит числом,
    // разбирать её умеет только изТаблицы
    сдаты,
    сдатыИзГлубины: датаВГлубине && !датойПохоже(р[4]),
    последний: отметка,
    // «Последний запуск» отмечает попытку, а не успех: отметитьЗапуск пишет её
    // и при ошибке. Для периода нужен именно успех, иначе упавший прогон сдвинет
    // окно вперёд и невыгруженные заказы будут потеряны навсегда.
    // Чтобы метка успеха пережила ошибку, она дописывается в текст «Результата».
    успешный: успехИз(результат, отметка),
  };
}

export async function задачи() {
  const строки = await читать(
    `'${ЛИСТ}'!A${НАСТРОЙКИ.задачиОт}:G${НАСТРОЙКИ.задачиДо}`,
  );
  const итог = [];
  строки.forEach((р, i) => {
    const имя = String(р[0] ?? '').trim();
    // строки не из списка задач пропускаем: на листе рядом могут быть другие блоки.
    // Но включённую строку с непохожим именем стоит назвать: опечатка в названии
    // задачи в книге неотличима от «срок ещё не подошёл»
    if (!ЗАДАЧИ.includes(имя)) {
      if (имя && Number(р[1]) === 1) {
        console.warn(`строка ${НАСТРОЙКИ.задачиОт + i}: «${имя}» включена, но такой задачи нет`);
      }
      return;
    }
    итог.push(задачаИзСтроки(р, НАСТРОЙКИ.задачиОт + i));
  });
  return итог;
}

/**
 * Настоящая дата, а не глубина. Отличаем по году: «31» в колонке глубины —
 * это серийный номер 31 января 1900 года, а не 2026-й.
 */
function датойПохоже(значение) {
  const д = изТаблицы(значение);
  return !!д && д.getFullYear() >= 2015;
}

/** «Глубина дней» бывает записана текстом: «31 д», «31 день». Берём ведущее число. */
function глубинаИз(значение) {
  if (typeof значение === 'number' && Number.isFinite(значение)) {
    return значение > 0 ? Math.floor(значение) : 0;
  }
  const м = String(значение ?? '').trim().match(/^(\d+)/);
  return м ? Number(м[1]) : 0;
}

/** Хвост «| успех: ДД.ММ.ГГГГ ЧЧ:ММ:СС», которым метка успеха переносится через ошибки. */
const МЕТКА = /\|\s*успех:\s*(\d{1,2}\.\d{1,2}\.\d{4} \d{1,2}:\d{2}:\d{2})/;

function успехИз(результат, отметка) {
  if (!результат.startsWith('ОШИБКА')) return отметка;
  const м = результат.match(МЕТКА);
  return м ? изТаблицы(м[1]) : null;
}

 * значит заново залить всю историю с этой даты.

 * иначе заказы за простой были бы потеряны навсегда.

  // никогда. Проверка стоит до обеих веток — в заливке дата берётся как есть
  const негодна = (с) => с.getTime() > по.getTime();
  const отказ = (с) => new Error(`«С даты» ${вТаблицу(с).slice(0, 10)} позже сегодняшнего дня`);

  if (граница && !знак) {

    return { с: граница, по, режим: 'заливка', историяГлубжеОкна: false };
  }

  let с = new Date(по.getTime() - дней * 24 * 3600 * 1000);
  if (знак && знак.getTime() - ЗАПАС < с.getTime()) с = new Date(знак.getTime() - ЗАПАС);

  /**
   * Заголовки дозаполняем ровно в пустые клетки первой строки и отдельными
   * диапазонами: занятую клетку не трогаем вовсе — там может стоять своя
   * формулировка или формула, а переписать её значением значило бы её потерять.
   */
  const правкиШапки = (шапка) => {
    const пустые = схема.map((_, i) => i).filter((i) => !String(шапка[i] ?? '').trim());
    return вДиапазоны(пустые).map((к) => ({
      range: `'${лист}'!${колонка(к[0] + 1)}1`,
      values: [к.map((i) => схема[i])],
    }));
  };

  if (СНИМКИ.has(лист)) {
    if (!строки.length) {
      // очистка снимка идёт до записи, поэтому пустой ответ стёр бы весь лист
      // и отметился бы в книге как успешный прогон
      throw new Error(`«${лист}»: пришло 0 строк, снимок не перезаписан — `
        + 'прежние остатки в книге сохранены');
    }
    // сначала пишем, потом убираем хвост: обрыв между шагами оставит лист
    // с данными, а не пустым. Границу берём по числу новых строк, а не по
    // колонке A: в остатках ВБ там штрихкод, а он бывает пустым
    const шапка = (await читать(`'${лист}'!A1:${буква}1`))[0] || [];
    const правки = правкиШапки(шапка);
    await писать(`'${лист}'!A2`, строки);
    if (правки.length) await писатьПачкой(правки);
    await очистить(лист, ширина, ПРЕДЕЛ, строки.length + 2);
    return {
      всего: строки.length, новых: строки.length, изменено: строки.length,
      заметка: правки.length ? ', шапка проставлена' : '',
    };
  }

  const ключи = КЛЮЧИ[лист] || [0];
  // ключ считаем тем же приведением, что и сравнение строк: книга возвращает
  // значения уже по-своему — штрихкод числом, «0468…» без ведущего нуля, — и
  // сырое сравнение не нашло бы такую строку в указателе, дописывая её заново
  // при каждом прогоне, пока лист не упрётся в предел
  const ключОт = (строка) => ключи.map((j) => сравнимо(строка[j])).join('|');
  // читаем с первой строки: шапка приезжает тем же запросом, лишних не надо
  const всё = await читать(`'${лист}'!A1:${буква}${ПРЕДЕЛ}`);
  const шапка = всё[0] || [];
  const было = всё.slice(1);
  const индекс = new Map();
  было.forEach((строка, i) => {
    const к = ключОт(строка);
    if (к.replace(/\|/g, '')) индекс.set(к, i);
  });

  if (было.length >= ПРЕДЕЛ - 1) {
    // дальше по листу есть строки, которых мы не видели: их ключи не найдутся,
    // а дозапись ляжет поверх них. Лучше честно упасть, чем испортить историю
    throw new Error(`лист «${лист}» дочитан до предела ${ПРЕДЕЛ} строк: `
      + 'перенесите старые заказы в архивный лист, дозапись остановлена');
  }

  const итог = было.map((с) => [...с]);
  const правленые = new Set();
  let новых = 0;
  let безКлюча = 0;
  for (const строка of строки) {
    const к = ключОт(строка);
    // строка без ключа не находится при следующем прогоне и дозаписывалась бы
    // заново каждый раз, пока лист не упрётся в предел
    if (!к.replace(/\|/g, '')) { безКлюча += 1; continue; }
    if (индекс.has(к)) {
      const i = индекс.get(к);
      if (одинаковы(итог[i], строка, ширина)) continue;
      итог[i] = строка;
      if (i < было.length) правленые.add(i);
    } else {
      индекс.set(к, итог.length);
      итог.push(строка);
      новых += 1;
    }
  }

  if (итог.length + 1 > ПРЕДЕЛ) {
    // дозапись ушла бы ниже окна чтения, и следующий прогон этих строк не увидит
    throw new Error(`«${лист}»: со свежими строками выходит ${итог.length} при пределе `
      + `${ПРЕДЕЛ} — перенесите старые заказы в архивный лист, запись остановлена`);
  }

  let заметка = '';
  if (итог.length + 1 > ПРЕДЕЛ * 0.9) {
    заметка = `, строк ${итог.length} при пределе ${ПРЕДЕЛ} — пора в архив`;
    console.warn(`[${лист}]${заметка}`);
  }
  if (безКлюча) заметка += `, без ключа пропущено ${безКлюча}`;

  // в книгу уходят ТОЛЬКО наши строки. Отправить обратно прочитанное из книги
  // нельзя: апостроф-пометка в ответ не возвращается, и правильный текстовый
  // артикул при обратной записи снова разобрался бы как дата, а формула
  // в колонке схемы затёрлась бы своим же вычисленным значением
  const данные = вДиапазоны(правленые).map((к) => ({
    range: `'${лист}'!A${к[0] + 2}`,
    values: к.map((i) => итог[i]),
  }));
  if (новых) {
    данные.push({ range: `'${лист}'!A${было.length + 2}`, values: итог.slice(было.length) });
  }
  const правкиШ = правкиШапки(шапка);
  данные.push(...правкиШ);
  if (правкиШ.length) заметка += ', шапка проставлена';

  for (let i = 0; i < данные.length; i += ДИАПАЗОНОВ_ЗА_РАЗ) {
    await писатьПачкой(данные.slice(i, i + ДИАПАЗОНОВ_ЗА_РАЗ));
  }

  return { всего: итог.length, новых, изменено: правленые.size, заметка };
}

/**
 * Списки отправлений отдают не больше сотни за раз: на 1000 метод отвечает
 * 400 «value must be inside range (0, 100]».
 *
 * Листать их `offset`-ом нельзя. У методов нового поколения такого поля нет,
 * а неизвестные поля метод молча выбрасывает — поэтому запрос выглядел рабочим,
 * отвечал 200 и на любой странице отдавал одну и ту же первую сотню.
 *
 * Обход идёт лестницей: следующая ступень включается сама, когда предыдущая
 * перестаёт двигать выборку.
 *   1. курсор метода;
 *   2. рез окна по дате последнего прочитанного отправления;
 *   3. делёж окна пополам.
 * Нижние две ступени опираются только на фильтр по датам, который заведомо
 * работает — им всё равно, чем именно метод листается и листается ли вообще.
 */
const ПАЧКА_ОТПРАВЛЕНИЙ = 100;
const ЗАПРОСОВ_МАКСИМУМ = 500;
// период длиннее года метод не берёт, а первичная заливка бывает и глубже.
// Режем с запасом: лишнее окно стоит один запрос
const ОКНО_ОТПРАВЛЕНИЙ = 180 * 24 * 3600 * 1000;
// пауза между запросами — вежливость к живому API; для подставного адреса не нужна
// ноль только для подставного адреса в пробе: выставь кто-нибудь OZON_API_URL
// на боевой хост — и паузы исчезли бы разом у отправлений и возвратов
const ПАУЗА_OZON = подставной(process.env.OZON_API_URL) ? 0 : 300;

/** Одна страница. Поля запроса — только те, что у метода есть. */
async function страницаОтправлений(метод, с, по, курсор) {
  const тело = {
    limit: ПАЧКА_ОТПРАВЛЕНИЙ,
    filter: { since: iso(с), to: iso(по) },
    with: { analytics_data: true, financial_data: true },
    sort_dir: 'ASC',
  };
  if (курсор) тело.cursor = курсор;

  await сон(ПАУЗА_OZON);
  const ответ = await ozon(метод, тело);
  const т = ответ.result || ответ;
  const список = Array.isArray(т) ? т : (т.postings || null);
  if (!Array.isArray(список)) {
    // пустое тело, 204 или ответ чужой формы. Раньше это молча считалось
    // концом окна, то есть потеря данных выглядела как успешный прогон
    throw new Error(`метод ответил без списка отправлений на окне с ${вТаблицу(с)}`);
  }
  return {
    список,
    курсор: Array.isArray(т) ? '' : String(т.cursor || т.next_cursor || т.last_id || ''),
    естьЕщё: Array.isArray(т) ? null : т.has_next,
  };
}

const датаОтправления = (о) => new Date(о.created_at || о.in_process_at);

/**
 * Где резать окно. Годится любая точка строго внутри него, поэтому берём дату
 * последнего прочитанного отправления, а если она не внутри — середину.
 *
 * Двигать НАЧАЛО окна по дате страницы нельзя: это молчаливая ставка на то, что
 * метод сортирует ровно по тому полю, по которому считает период. Разойдись они —
 * начало перескочит через хвост сортировки, и отправления пропадут без единого
 * слова. Рез конца такой ставки не делает: две половины покрывают окно целиком
 * при любом месте реза.
 */
function точкаРеза(список, с, по) {
  const даты = список.map(датаОтправления)
    .filter((д) => !Number.isNaN(д.getTime())).map((д) => д.getTime());
  const макс = даты.length ? Math.max(...даты) : null;
  if (макс !== null && макс > с.getTime() && макс < по.getTime()) {
    return { дата: new Date(макс), способ: 'дата' };
  }
  const середина = Math.floor((с.getTime() + по.getTime()) / 2);
  if (середина > с.getTime() && середина < по.getTime()) {
    return { дата: new Date(середина), способ: 'делёж' };
  }
  return null;
}

async function обходОкна(метод, с, по, ст) {
  let курсор = '';
  let последняя = null;

  for (;;) {
    if (ст.запросов >= ЗАПРОСОВ_МАКСИМУМ) {
      throw new Error(`прочитано ${ст.собрано.length} отправлений и упёрлись в предел `
        + `${ЗАПРОСОВ_МАКСИМУМ} запросов — сузьте период`);
    }
    ст.запросов += 1;
    const стр = await страницаОтправлений(метод, с, по, курсор);
    последняя = стр.список;

    let свежих = 0;
    for (const о of стр.список) {
      const ключ = String(о.posting_number || '');
      if (ключ && ст.виденные.has(ключ)) continue;
      if (ключ) ст.виденные.add(ключ);
      ст.собрано.push(о);
      свежих += 1;
    }

    // короткая страница — конец окна, но только если метод не сказал прямо, что
    // есть ещё: недолить страницу он вправе, и вера в её длину обрезала бы
    // выборку молча. Сам `has_next: false` конец окна закрывает при любой длине
    if (стр.список.length < ПАЧКА_ОТПРАВЛЕНИЙ && стр.естьЕщё !== true) return;
    if (стр.естьЕщё === false) return;
    if (стр.курсор && стр.курсор !== курсор && свежих) {
      курсор = стр.курсор;
      ст.способы.add('курсор');
      continue;
    }
    break;   // курсор не двигает выборку — дальше идём по датам
  }

  // вся полная страница пришла одним мгновением: никакой рез её не разъединит,
  // а курсора нет. Ловим сразу, иначе обход сорок раз поделит окно пополам
  // и упрётся в предел запросов, сообщив совсем не о том
  const мгновения = new Set(последняя.map((о) => датаОтправления(о).getTime()));
  if (мгновения.size === 1) {
    throw new Error(`окно на ${вТаблицу(последняя[0] && датаОтправления(последняя[0]))} `
      + `уже не делится, а ${ПАЧКА_ОТПРАВЛЕНИЙ} отправлений в нём не кончаются`);
  }

  const рез = точкаРеза(последняя, с, по);
  if (!рез) {
    throw new Error(`окно на ${вТаблицу(с)} уже не делится, `
      + `а ${ПАЧКА_ОТПРАВЛЕНИЙ} отправлений в нём не кончаются`);
  }
  ст.способы.add(рез.способ);
  if (!ст.способы.has('курсор')) console.warn(`[${метод}] курсор не двигает выборку, дальше идём по дате`);

  await обходОкна(метод, с, рез.дата, ст);
  // правая половина начинается миллисекундой РАНЬШЕ реза: понимание границ
  // у метода неизвестно, и пойми он строго обе — отправление ровно на шве
  // выпало бы из обеих половин. Перекрытие снимет указатель прочитанных
  await обходОкна(метод, new Date(рез.дата.getTime() - 1), по, ст);
}

export async function ozonОтправления(метод, задача, кластер) {
  const { с, по } = период(задача);
  const ст = { виденные: new Set(), собрано: [], запросов: 0, способы: new Set() };
  // длинный период сразу режем на окна по пределу метода, не дожидаясь от него 400
  for (let край = с; ;) {
    const конец = new Date(Math.min(край.getTime() + ОКНО_ОТПРАВЛЕНИЙ, по.getTime()));
    await обходОкна(метод, край, конец, ст);
    if (конец.getTime() >= по.getTime()) break;
    // шов перекрываем на миллисекунду — по той же причине, что и при резе окна
    край = new Date(конец.getTime() - 1);
  }

  const отметка = вТаблицу(new Date());
  const строки = ст.собрано.flatMap((о) => {
    const а = о.analytics_data || {};
    const ф = о.financial_data || {};
    return (о.products || []).map((т) => [
      о.posting_number, о.order_number || '-',
      вТаблицу(датаОтправления(о)),
      текстом(т.offer_id) || '-', String(т.sku), число(т.quantity), число(т.price),
      о.status || '-',
      // кластер приходит только в financial_data, а склад у FBS зовётся warehouse,
      // а не warehouse_name: перебираем оба, чтобы не зависеть от схемы метода
      кластер
        ? (ф.cluster_to || а.cluster_to || а.warehouse_name || а.warehouse || '-')
        : (а.warehouse_name || а.warehouse || '-'),
      а.region || '-', отметка,
    ]);
  });

  const порядок = ['курсор', 'дата', 'делёж'].filter((с_) => ст.способы.has(с_));
  return {
    строки,
    сводка: `отправлений ${ст.собрано.length}, запросов ${ст.запросов}, `
          + `листали: ${порядок.join(' + ') || 'одна страница'}`,
  };
}

}

/**
 * Возвраты FBO и FBS приходят одним методом, поэтому лист под них один, а схема
 * конкретного возврата пишется в колонку «Схема».
 *
 * Период берётся из «С даты» как есть: весь отрезок от неё до сегодня, каждый
 * прогон заново. Глубина и водяной знак тут не участвуют — метод принимает
 * только один фильтр-дату за запрос, и дробить отрезок нечем. Пустая «С даты»
 * значит «все возвраты, какие есть».
 *
 * Фильтруем по дате возврата: это та же дата, что уходит в колонку
 * «Дата возврата», поэтому период в книге совпадает с тем, что видно в листе.
 *
 * Листается `last_id` — идентификатор последнего возврата на странице.
 */
const ВОЗВРАТЫ = '/v1/returns/list';
const ЛИСТ_ВОЗВРАТОВ = 'Возвраты OZON';
const ПАЧКА_ВОЗВРАТОВ = 500;      // предел метода
const СТРАНИЦ_ВОЗВРАТОВ = 200;    // сто тысяч возвратов

/** Суммы приходят объектом {currency_code, price}, причём price — строкой. */
const деньги = (з) => число(з && з.price);

/** Сбор и раскладка по колонкам, без записи в книгу: так его проверяет проба. */
export async function ozonВозвратыСтроки(задача) {
  const с = границаИстории(задача || {});
  const фильтр = с
    ? { logistic_return_date: { time_from: iso(с), time_to: iso(new Date()) } }
    : null;

  const собрано = [];
  const виденные = new Set();
  let last = 0;
  let дочитано = false;

  for (let n = 0; n < СТРАНИЦ_ВОЗВРАТОВ && !дочитано; n += 1) {
    await сон(ПАУЗА_OZON);
    const тело = { limit: ПАЧКА_ВОЗВРАТОВ, last_id: last };
    if (фильтр) тело.filter = фильтр;
    const ответ = await ozon(ВОЗВРАТЫ, тело);
    const ответТело = ответ.result || ответ;
    // пустое тело, 204 или ответ чужой формы. Молча счесть это концом списка
    // значило бы записать «возвратов 0» как успешный прогон
    if (!Array.isArray(ответТело.returns)) {
      throw new Error(`возвраты: метод ответил без списка на странице ${n + 1}`);
    }
    const пачка = ответТело.returns;

    for (const в of пачка) {
      const ключ = String(в.id ?? '');
      if (ключ && виденные.has(ключ)) continue;
      if (ключ) виденные.add(ключ);
      собрано.push(в);
    }

    if (!пачка.length || ответТело.has_next === false) { дочитано = true; break; }
    if (ответТело.has_next === undefined && пачка.length < ПАЧКА_ВОЗВРАТОВ) {
      дочитано = true; break;
    }

    const хвост = Number(пачка[пачка.length - 1].id);
    // та же беда, что была у отправлений: курсор, который не двигает выборку,
    // выглядит как бесконечный список одних и тех же записей
    if (!Number.isSafeInteger(хвост) || хвост === last) {
      throw new Error(`возвраты: last_id не двигает выборку, прочитано ${собрано.length}`);
    }
    last = хвост;
  }
  if (!дочитано) {
    throw new Error(`возвраты: прочитано ${собрано.length} и упёрлись `
      + `в предел ${СТРАНИЦ_ВОЗВРАТОВ} страниц`);
  }

  const отметка = вТаблицу(new Date());
  const строки = собрано.map((в) => {
    const т = в.product || {};
    const вид = в.visual || {};
    const лог = в.logistic || {};
    return [
      String(в.id ?? ''), в.posting_number || '-', в.order_number || '-',
      в.schema || '-',
      текстом(т.offer_id) || '-', String(т.sku ?? ''), текстом(т.name) || '-',
      число(т.quantity), деньги(т.price),
      в.return_reason_name || '-', в.type || '-',
      вид.status?.display_name || '-', вТаблицу(вид.change_moment),
      вТаблицу(лог.return_date),
      в.place?.name || '-', текстом(лог.barcode) || '-',
      отметка,
    ];
  });

  // фильтр мог быть отвергнут или понят иначе: метод принимает лишь одну дату
  // за запрос, и молча вернуть всё подряд ему ничего не стоит. Считаем, сколько
  // пришло вне запрошенного отрезка, чтобы это не осталось незамеченным
  const мимо = с ? собрано.filter((в) => {
    const сырое = в.logistic?.return_date;
    // new Date(null) это 1 января 1970 года, и без этой проверки каждый возврат
    // без даты поднимал бы ложную тревогу «фильтр не применён»
    if (typeof сырое !== 'string' || !сырое) return false;
    const д = new Date(сырое);
    return !Number.isNaN(д.getTime()) && д.getTime() < с.getTime();

/** Яндекс отдаёт ДД-ММ-ГГГГ — конструктор Date такое читает неверно. */
function изЯМ(строка) {
  if (!строка) return null;
  const м = String(строка).match(/^(\d{2})-(\d{2})-(\d{4})(?:\s+(\d{2}):(\d{2}):(\d{2}))?$/);
  if (!м) return new Date(строка);

/**
 * Остатки приходят разложенными по типам, и типы **пересекаются**:
 *
 *   `AVAILABLE` («Доступный к заказу») — товар, доступный для продажи;
 *   `FREEZE` — товар, зарезервированный под заказы;
 *   `FIT` («Годный») — товар, доступный для продажи ИЛИ уже зарезервированный,
 *   то есть `AVAILABLE` + `FREEZE`.
 *
 * Складывать `AVAILABLE` с `FIT`, как если бы это были два имени одного и того
 * же, нельзя — свободный остаток посчитается дважды. Ровно на этом лист показывал
 * двойные остатки и по FBO, и по FBS: при десяти штуках без резерва и в «Доступно»,
 * и в «Всего» стояло двадцать.
 *
 * Остальные типы — `DEFECT`, `EXPIRED`, `QUARANTINE`, `UTILIZATION` — лежат рядом
 * с годным, а не внутри него, поэтому в «Всего» идут как есть.
 */
const ВНУТРИ_ГОДНОГО = new Set(['AVAILABLE', 'FREEZE']);

function остатокТовара(товар) {
  const поТипу = new Map();
  for (const о of товар.stocks || []) {
    поТипу.set(о.type, (поТипу.get(о.type) || 0) + число(о.count));
  }
  const заморожено = поТипу.get('FREEZE') || 0;
  // не дал «годный» — собираем его из частей; не дал «доступный» — вычитаем резерв
  const годный = поТипу.has('FIT')
    ? поТипу.get('FIT')
    : (поТипу.get('AVAILABLE') || 0) + заморожено;
  const доступно = поТипу.has('AVAILABLE')
    ? поТипу.get('AVAILABLE')
    : Math.max(0, годный - заморожено);
  let прочее = 0;
  for (const [тип, к] of поТипу) {
    if (тип !== 'FIT' && !ВНУТРИ_ГОДНОГО.has(тип)) прочее += к;
  }
  return { доступно, заморожено, брак: поТипу.get('DEFECT') || 0, всего: годный + прочее };
}

 * `offset` и `dir`, на которых сервис однажды уже обжёгся: живой метод их молча
 * выбрасывает, отвечает 200, и запрос выглядит рабочим, не будучи им.

const ПОЛЯ = new Set(['limit', 'filter', 'with', 'sort_dir', 'cursor', 'translit']);
const ПОЛЯ_ФИЛЬТРА = new Set(['since', 'to', 'status']);

function поддельныйOzon({
  отправления, курсорРаботает = false, полеСортировки = 'created_at',
  недолив = 0, пределПериода = 0,
}) {
  const запросы = [];
  const сервер = http.createServer((req, res) => {
    let сырое = '';
    req.on('data', (ч) => { сырое += ч; });
    req.on('end', () => {
      const тело = JSON.parse(сырое || '{}');
      запросы.push(тело);
      const отказ = (текст) => {
        res.writeHead(400, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ code: 3, message: текст }));
      };

      for (const к of Object.keys(тело)) {
        if (!ПОЛЯ.has(к)) return отказ(`unknown field "${к}"`);
      }
      for (const к of Object.keys(тело.filter || {})) {
        if (!ПОЛЯ_ФИЛЬТРА.has(к)) return отказ(`unknown filter field "${к}"`);
      }
      if (!(тело.limit > 0 && тело.limit <= 100)) {
        return отказ('invalid Limit: value must be inside range (0, 100]');
      }

      const с = new Date(тело.filter.since).getTime();
      const по = new Date(тело.filter.to).getTime();
      // живой метод не берёт период длиннее года
      if (пределПериода && по - с > пределПериода) return отказ('PERIOD_IS_TOO_LONG');
      // фильтруем по created_at, а сортируем, возможно, по другому полю:
      // так проверяется, что обход не двигает начало окна по дате страницы
      const подходят = отправления
        .filter((о) => { const д = new Date(о.created_at).getTime(); return д >= с && д <= по; })
        .sort((а, б) => String(а[полеСортировки]).localeCompare(String(б[полеСортировки])));

      const от = курсорРаботает && тело.cursor ? Number(тело.cursor) : 0;
      // недолив: метод вправе отдать меньше запрошенного, не соврав про has_next
      const пачка = подходят.slice(от, от + Math.min(тело.limit, недолив || тело.limit));
      const тело_ = { postings: пачка };
      if (курсорРаботает) {
        тело_.cursor = String(от + пачка.length);
        тело_.has_next = от + пачка.length < подходят.length;
      }
      res.writeHead(200, { 'content-type': 'application/json' });

    });

  for (const [целиком, имя, номер] of текст.matchAll(ССЫЛКА)) {
    всего += 1;
    if (!кэш.has(имя)) кэш.set(имя, await найти(имя));
    const файл = кэш.get(имя);
    if (!файл) {
      console.log(`  ${док}: ${целиком} — такого файла нет`);
      плохо += 1;
      continue;
    }
    const строки = файл.текст.split('\n');
    const n = Number(номер);
    if (n < 1 || n > строки.length) {
      console.log(`  ${док}: ${целиком} — за пределами файла (строк ${строки.length})`);
      плохо += 1;
    } else if (!строки[n - 1].trim()) {
      console.log(`  ${док}: ${целиком} — пустая строка`);
      плохо += 1;
    }
  }
}

console.log(`ссылок проверено ${всего}, битых ${плохо}`);
